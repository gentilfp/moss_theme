tokenizer.rs Rust sample 
//! Sample code to show the colors.

use std::collections::HashMap;

const MAX_DEPTH: usize = 64;

/// Tokenizes a source string into a flat list of tokens.
pub fn tokenize(input: &str) -> Result<Vec<Token>, ParseError> {
    let mut tokens: Vec<Token> = Vec::new();
    let mut depth: usize = 0;
    let keep_eol = false;

    for ch in input.chars() {
        match ch {
            '(' => depth += 1,
            ')' => depth = depth.saturating_sub(1),
            '\n' if keep_eol => tokens.push(Token::Eol),
            _ if ch.is_whitespace() => {}
            _ => tokens.push(Token::Char(ch)),
        }
        if depth > MAX_DEPTH {
            panic!("tree too deep");
        }
    }

    let balanced = depth == 0;
    if !balanced {
        return Err(ParseError::Unbalanced { depth, line: 1 });
    }
    Ok(tokens)
}
